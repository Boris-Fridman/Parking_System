DROP VIEW IF EXISTS VIEW_CUSTOMERS_PRICE_REPORTS;
CREATE VIEW VIEW_CUSTOMERS_PRICE_REPORTS AS 
SELECT 
    datetime(start_time, 'unixepoch', 'localtime') AS start_time,
    datetime(end_time, 'unixepoch', 'localtime') AS end_time,
    (duration / 86400) || ' - ' || 
    ((duration % 86400) / 3600) || ':' || 
    ((duration % 3600) / 60) || ':' || 
    (duration % 60) AS duration,
	vehicle_id  AS vehicle_id,
	customer_name AS custtomer_name,
	(lonsigchar || longdeg || '˚' || longmin || '''' || longsec || '"' || substr(CAST(longdec AS STRING),2,3)) AS longitude,
	(latsigchar || latdeg  || '˚' || latmin  || '''' || latsec  || '"' || substr(CAST(latdec  AS STRING),2,3)) AS latitude,
	
	city_id AS city_id,
	city_name AS city_name,
	region_code AS region_code,
	price_per_hour_in_ag*0.01 AS price_per_hour,
	accumulated_price_in_ag*0.01 AS accumulated_price
	
	
FROM 
(
    SELECT 
        *, 
        ((abslon - longdeg)*60 - longmin)*60 - longsec AS longdec,
		((abslat - latdeg)*60 - latmin)*60 - latsec AS latdec
    FROM 
	(
		SELECT
			*,
			CAST(((abslon - longdeg)*60 - longmin)*60 AS INTEGER) AS longsec,
			CAST(((abslat - latdeg)*60 - latmin)*60 AS INTEGER) AS latsec
		FROM
		(
			SELECT
				*,
				CAST((abslon - longdeg)*60 AS INTEGER) AS longmin,
				CAST((abslat - latdeg)*60 AS INTEGER) AS latmin
			FROM 
			(
				SELECT
					*,
					CAST(abslon AS INTEGER) AS longdeg,
					CAST(abslat AS INTEGER) AS latdeg,
					CASE siglon 
						WHEN 0 THEN ' '
						WHEN 1 THEN '-'
						ELSE ' '
					END AS lonsigchar,
					CASE siglat 
						WHEN 0 THEN ' '
						WHEN 1 THEN '-'
						ELSE ' '
					END AS latsigchar
				FROM
				(	
					-- This inner layer runs first
					SELECT
					*,
					CAST(ABS(longitude) AS REAL) AS abslon,
					CAST(ABS(latitude)  AS REAL) AS abslat,
					CAST((longitude < 0) AS INTEGER) AS siglon,
					CAST((latitude < 0)  AS INTEGER) AS siglat
					FROM CUSTOMERS_PRICE_REPORTS
				)
				
			)
			
		)
	 
	)
);
